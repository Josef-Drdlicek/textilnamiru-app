<?php

namespace App\Http\Controllers\Web;

use App\Domain\Pricing\ItemPricer;
use App\Domain\Pricing\PublicPresets;
use App\Enums\ItemType;
use App\Http\Controllers\Controller;
use Illuminate\Support\Facades\Cache;
use Illuminate\View\View;

class PageController extends Controller
{
    public function home(): View
    {
        return view('web.home');
    }

    public function sortiment(): View
    {
        return view('web.sortiment', [
            'pricesFrom' => $this->pricesFrom(),
        ]);
    }

    public function kalkulacka(): View
    {
        return view('web.kalkulacka');
    }

    public function realizace(): View
    {
        return view('web.realizace');
    }

    public function vypinanyPodhled(): View
    {
        return view('web.vypinany-podhled');
    }

    public function stiniciTechnika(): View
    {
        return view('web.stinici-technika');
    }

    public function b2b(): View
    {
        return view('web.b2b');
    }

    public function privacy(): View
    {
        return view('web.gdpr');
    }

    /**
     * Katalogová „cena od" pro každý typ — s DPH, zaokrouhlená na desítky
     * korun nahoru. Ceník se mění zřídka, hodinová cache bohatě stačí.
     *
     * @return array<string, int>
     */
    private function pricesFrom(): array
    {
        return Cache::remember('web.prices-from', now()->addHour(), function (): array {
            $pricer = app(ItemPricer::class);

            try {
                $vatRate = $pricer->vatRate();
            } catch (\DomainException $e) {
                report($e);

                return [];
            }

            $prices = [];

            foreach (ItemType::cases() as $type) {
                // Díra v ceníku nesmí shodit celý web — dlaždice bez ceny
                // spadne na chip „Na poptávku".
                try {
                    $price = $pricer->price($type, PublicPresets::referenceParameters($type));
                    $prices[$type->value] = (int) (ceil($price->unitPriceWithVat($vatRate) / 10) * 10);
                } catch (\DomainException $e) {
                    report($e);
                }
            }

            return $prices;
        });
    }
}

<?php

namespace App\Livewire\Web;

use App\Domain\Pricing\ItemPricer;
use App\Domain\Pricing\PublicPresets;
use App\Enums\ItemType;
use Illuminate\Support\Facades\Validator;
use Illuminate\Validation\Rule;
use Illuminate\View\View;
use Livewire\Component;

/**
 * Veřejná orientační kalkulačka. Vystavuje jen laické parametry (rozměry,
 * počet, nařasení, výplň) — odborné parametry doplňuje PublicPresets, takže
 * čísla sedí s katalogovou „cenou od". Cena se počítá výhradně na serveru.
 */
class PriceEstimator extends Component
{
    public string $type = 'curtain';

    public ?float $width = null;

    public ?float $height = null;

    public ?int $count = 1;

    /** Klíč z PublicPresets::RUFFLE_OPTIONS (jen záclony a závěsy). */
    public string $ruffle = 'standard';

    /** Včetně výplně (jen polštáře a sedáky). */
    public bool $stuffing = false;

    // Výstupy kalkulace — pouze pro zobrazení; snapshot pro poptávku vzniká
    // vždy znovu na serveru v sendToInquiry().
    public ?float $unitPriceVat = null;

    public ?float $totalPrice = null;

    public ?float $totalPriceVat = null;

    /** @var list<array{label: string, amount: float}> */
    public array $priceLines = [];

    public function mount(): void
    {
        $requested = ItemType::tryFrom((string) request()->query('typ'));

        if ($requested !== null) {
            $this->type = $requested->value;
        }

        $this->prefillDimensions();
        $this->recalculate();
    }

    public function itemType(): ItemType
    {
        return ItemType::from($this->type);
    }

    public function updated(string $property): void
    {
        $this->recalculate();

        if ($property !== 'type') {
            $this->validateOnly($property);
        }
    }

    public function updatedType(): void
    {
        $this->resetValidation();
        $this->prefillDimensions();
        $this->recalculate();
    }

    /**
     * Uloží serverový snapshot konfigurace do session a přejde na poptávku.
     */
    public function sendToInquiry()
    {
        $this->validate();

        $pricer = app(ItemPricer::class);
        $type = $this->itemType();
        $parameters = PublicPresets::estimatorParameters($type, $this->estimatorInput());

        try {
            $result = $pricer->price($type, $parameters);
        } catch (\DomainException) {
            $this->addError('width', 'Tuto konfiguraci neumíme nacenit online — poptejte nás prosím napřímo.');

            return null;
        }

        $vatRate = $pricer->vatRate();
        $count = (int) $this->count;

        session()->put('inquiry.configuration', [
            'type' => $type->value,
            'parameters' => $parameters->toArray(),
            'count' => $count,
            'unit_price' => $result->unitPrice,
            'unit_price_vat' => $result->unitPriceWithVat($vatRate),
            'total_price_vat' => $result->totalPriceWithVat($vatRate, $count),
            'vat_rate' => $vatRate,
            'price_lines' => $result->linesToArray(),
            'summary' => $this->summary(),
            'calculated_at' => now()->toIso8601String(),
        ]);

        return $this->redirectRoute('web.inquiry.create');
    }

    public function isFabricType(): bool
    {
        return in_array($this->itemType(), [ItemType::Curtain, ItemType::Drape], true);
    }

    public function hasStuffingOption(): bool
    {
        return in_array($this->itemType(), [ItemType::Pillow, ItemType::Cushion], true);
    }

    /**
     * @return array<string, list<mixed>>
     */
    protected function rules(): array
    {
        [$maxWidth, $maxHeight] = match ($this->itemType()) {
            ItemType::Curtain, ItemType::Drape => [2000, 500],
            ItemType::Blind => [300, 300],
            ItemType::Pillow, ItemType::Cushion => [100, 100],
        };

        return [
            'type' => ['required', Rule::enum(ItemType::class)],
            'width' => ['required', 'numeric', 'min:10', "max:{$maxWidth}"],
            'height' => ['required', 'numeric', 'min:10', "max:{$maxHeight}"],
            'count' => ['required', 'integer', 'min:1', 'max:999'],
            'ruffle' => ['required', Rule::in(array_keys(PublicPresets::RUFFLE_OPTIONS))],
            'stuffing' => ['boolean'],
        ];
    }

    /**
     * @return array<string, string>
     */
    protected function validationAttributes(): array
    {
        return [
            'width' => 'šířka',
            'height' => 'výška',
            'count' => 'počet kusů',
            'ruffle' => 'nařasení',
        ];
    }

    /**
     * Lidsky čitelný souhrn konfigurace pro poptávku a admin.
     */
    public function summary(): string
    {
        $parts = [
            $this->itemType()->label(),
            sprintf('%s × %s cm', rtrim(rtrim(number_format((float) $this->width, 1, ',', ' '), '0'), ','), rtrim(rtrim(number_format((float) $this->height, 1, ',', ' '), '0'), ',')),
            $this->count.' ks',
        ];

        if ($this->isFabricType()) {
            $parts[] = mb_strtolower(PublicPresets::RUFFLE_OPTIONS[$this->ruffle]['label']).' nařasení';
        }

        if ($this->hasStuffingOption()) {
            $parts[] = $this->stuffing ? 'včetně výplně' : 'bez výplně';
        }

        return implode(' · ', $parts);
    }

    /**
     * @return array{width: float, height: float, ratio: float, stuffing: bool}
     */
    protected function estimatorInput(): array
    {
        return [
            'width' => (float) $this->width,
            'height' => (float) $this->height,
            'ratio' => PublicPresets::RUFFLE_OPTIONS[$this->ruffle]['ratio'],
            'stuffing' => $this->stuffing,
        ];
    }

    protected function prefillDimensions(): void
    {
        $reference = PublicPresets::referenceParameters($this->itemType())->toArray();

        $this->width = $reference['width'];
        $this->height = $reference['height'];
        $this->count = 1;
        $this->ruffle = 'standard';
        $this->stuffing = false;
    }

    protected function recalculate(): void
    {
        if (! $this->hasValidInput()) {
            $this->clearPrice();

            return;
        }

        $pricer = app(ItemPricer::class);

        try {
            $result = $pricer->price(
                $this->itemType(),
                PublicPresets::estimatorParameters($this->itemType(), $this->estimatorInput()),
            );
            $vatRate = $pricer->vatRate();
        } catch (\DomainException) {
            $this->clearPrice();

            return;
        }

        $this->unitPriceVat = $result->unitPriceWithVat($vatRate);
        $this->totalPrice = $result->totalPrice((int) $this->count);
        $this->totalPriceVat = $result->totalPriceWithVat($vatRate, (int) $this->count);
        $this->priceLines = $result->linesToArray();
    }

    protected function clearPrice(): void
    {
        $this->unitPriceVat = null;
        $this->totalPrice = null;
        $this->totalPriceVat = null;
        $this->priceLines = [];
    }

    protected function hasValidInput(): bool
    {
        $rules = $this->rules();

        $data = [];

        foreach (array_keys($rules) as $field) {
            $data[$field] = $this->{$field};
        }

        return ! Validator::make($data, $rules)->fails();
    }

    public function render(): View
    {
        return view('livewire.web.price-estimator');
    }
}

<?php

namespace App\Domain\Pricing;

use App\Domain\Pricing\Parameters\BlindParameters;
use App\Domain\Pricing\Parameters\CurtainParameters;
use App\Domain\Pricing\Parameters\CushionParameters;
use App\Domain\Pricing\Parameters\DrapeParameters;
use App\Domain\Pricing\Parameters\ItemParameters;
use App\Domain\Pricing\Parameters\PillowParameters;
use App\Enums\ItemType;
use App\Enums\RightLeft;
use App\Enums\ZipperSide;

/**
 * Referenční konfigurace pro veřejný web. Jediné místo, kde se laický vstup
 * z kalkulačky (rozměry, nařasení, výplň) doplňuje o odborné parametry,
 * které běžný zákazník nezná — díky tomu „cena od" v katalogu i orientační
 * cena v kalkulačce počítají vždy nad stejnými předpoklady.
 */
final class PublicPresets
{
    /** Kód výchozí řasící stuhy (povolené poměry 1.5 / 1.85 / 2 / 2.5). */
    public const string DEFAULT_RIBBON = 'siroky70';

    /** Šířka role materiálu předpokládaná u záclon [cm] (voály se šijí z výšky role). */
    public const float CURTAIN_MATERIAL_WIDTH = 300.0;

    /** Šířka role materiálu předpokládaná u závěsů [cm] (dekorační látky). */
    public const float DRAPE_MATERIAL_WIDTH = 140.0;

    /** Veřejné volby nařasení: klíč => [popisek, poměr]. */
    public const array RUFFLE_OPTIONS = [
        'light' => ['label' => 'Jemné (1 : 1,5)', 'ratio' => 1.5],
        'standard' => ['label' => 'Standardní (1 : 2)', 'ratio' => 2.0],
        'rich' => ['label' => 'Bohaté (1 : 2,5)', 'ratio' => 2.5],
    ];

    /**
     * Parametry pro orientační kalkulačku — laický vstup doplněný o defaulty.
     *
     * @param  array{width: float, height: float, ratio?: float, stuffing?: bool}  $input
     */
    public static function estimatorParameters(ItemType $type, array $input): ItemParameters
    {
        return match ($type) {
            ItemType::Curtain => new CurtainParameters(
                width: $input['width'],
                height: $input['height'],
                materialWidth: self::CURTAIN_MATERIAL_WIDTH,
                ribbonCode: self::DEFAULT_RIBBON,
                ratio: $input['ratio'] ?? 2.0,
                kanyr: false,
                kanyrHeight: null,
                rightLeft: RightLeft::Right,
                corners: false,
                bottom: false,
                narasit: false,
                snails: false,
            ),
            ItemType::Drape => new DrapeParameters(
                width: $input['width'],
                height: $input['height'],
                materialWidth: self::DRAPE_MATERIAL_WIDTH,
                ribbonCode: self::DEFAULT_RIBBON,
                ratio: $input['ratio'] ?? 2.0,
                raport: null,
                folding: null,
                kanyr: false,
                kanyrHeight: null,
                rightLeft: RightLeft::Right,
                corners: false,
                bottom: false,
                narasit: false,
                snails: false,
            ),
            ItemType::Pillow => new PillowParameters(
                width: $input['width'],
                height: $input['height'],
                stuffing: $input['stuffing'] ?? false,
                zipper: ZipperSide::Short,
            ),
            ItemType::Cushion => new CushionParameters(
                width: $input['width'],
                height: $input['height'],
                depth: 5.0,
                stuffing: $input['stuffing'] ?? false,
                zipper: ZipperSide::Short,
            ),
            ItemType::Blind => new BlindParameters(
                width: $input['width'],
                height: $input['height'],
                mechanism: false,
                tunnels: true,
                finishing: true,
                cells: null,
                cellHeightTop: null,
                cellHeight: null,
                cellHeightBottom: null,
            ),
        };
    }

    /**
     * Nejmenší rozumná konfigurace typu — základ katalogové „ceny od".
     */
    public static function referenceParameters(ItemType $type): ItemParameters
    {
        return self::estimatorParameters($type, match ($type) {
            ItemType::Curtain => ['width' => 100.0, 'height' => 120.0],
            ItemType::Drape => ['width' => 140.0, 'height' => 250.0],
            ItemType::Pillow => ['width' => 40.0, 'height' => 40.0],
            ItemType::Cushion => ['width' => 40.0, 'height' => 40.0],
            ItemType::Blind => ['width' => 60.0, 'height' => 120.0],
        });
    }
}

use App\Domain\Pricing\Calculators\BlindPriceCalculator;
use App\Domain\Pricing\Calculators\CurtainPriceCalculator;
use App\Domain\Pricing\Calculators\CushionPriceCalculator;
use App\Domain\Pricing\Calculators\DrapePriceCalculator;
use App\Domain\Pricing\Calculators\PillowPriceCalculator;
use App\Domain\Pricing\Parameters\BlindParameters;
use App\Domain\Pricing\Parameters\CurtainParameters;
use App\Domain\Pricing\Parameters\CushionParameters;
use App\Domain\Pricing\Parameters\DrapeParameters;
use App\Domain\Pricing\Parameters\ItemParameters;
use App\Domain\Pricing\Parameters\PillowParameters;
use App\Domain\Pricing\PriceCalculator;
use App\Domain\Schema\Renderers\BlindSchemaRenderer;
use App\Domain\Schema\Renderers\CurtainSchemaRenderer;
use App\Domain\Schema\Renderers\CushionSchemaRenderer;
use App\Domain\Schema\Renderers\DrapeSchemaRenderer;
use App\Domain\Schema\Renderers\PillowSchemaRenderer;
use App\Domain\Schema\SchemaRenderer;
use App\Livewire\Configurators\AbstractConfigurator;
use App\Livewire\Configurators\BlindConfigurator;
use App\Livewire\Configurators\CurtainConfigurator;
use App\Livewire\Configurators\CushionConfigurator;
use App\Livewire\Configurators\DrapeConfigurator;
use App\Livewire\Configurators\PillowConfigurator;

enum ItemType: string
{
    case Curtain = 'curtain';
    case Drape = 'drape';
    case Pillow = 'pillow';
    case Cushion = 'cushion';
    case Blind = 'blind';

    /**
     * @return class-string<PriceCalculator>
     */
    public function calculatorClass(): string
    {
        return match ($this) {
            self::Curtain => CurtainPriceCalculator::class,
            self::Drape => DrapePriceCalculator::class,
            self::Pillow => PillowPriceCalculator::class,
            self::Cushion => CushionPriceCalculator::class,
            self::Blind => BlindPriceCalculator::class,
        };
    }

    /**
     * @return class-string<ItemParameters>
     */
    public function parametersClass(): string
    {
        return match ($this) {
            self::Curtain => CurtainParameters::class,
            self::Drape => DrapeParameters::class,
            self::Pillow => PillowParameters::class,
            self::Cushion => CushionParameters::class,
            self::Blind => BlindParameters::class,
        };
    }

    /**
     * @return class-string<AbstractConfigurator>
     */
    public function configuratorClass(): string
    {
        return match ($this) {
            self::Curtain => CurtainConfigurator::class,
            self::Drape => DrapeConfigurator::class,
            self::Pillow => PillowConfigurator::class,
            self::Cushion => CushionConfigurator::class,
            self::Blind => BlindConfigurator::class,
        };
    }

    /**
     * @return class-string<SchemaRenderer>
     */
    public function schemaRendererClass(): string
    {
        return match ($this) {
            self::Curtain => CurtainSchemaRenderer::class,
            self::Drape => DrapeSchemaRenderer::class,
            self::Pillow => PillowSchemaRenderer::class,
            self::Cushion => CushionSchemaRenderer::class,
            self::Blind => BlindSchemaRenderer::class,
        };
    }

    public function label(): string
    {
        return match ($this) {
            self::Curtain => 'Záclona',
            self::Drape => 'Závěs',
            self::Pillow => 'Polštář',
            self::Cushion => 'Sedák',
            self::Blind => 'Římská roleta',
        };
    }
}
